### 1.4 实现最简单的JNI函数

在 `rust/wgpu-mc-jni/src/lib.rs` 中：

```rust
use jni::objects::{JClass, JString};
use jni::JNIEnv;

/// 导出函数命名规则：Java_<完整类路径>_<方法名>
/// 对应 Java 类：com.dx12.D3D12Bridge
#[no_mangle]
pub unsafe extern "C" fn Java_com_dx12_D3D12Bridge_nativeInit(
    env: JNIEnv,
    _class: JClass,
) {
    // 初始化日志（方便调试）
    env_logger::init();
    log::info!("Rust JNI library loaded successfully!");
}

#[no_mangle]
pub unsafe extern "C" fn Java_com_dx12_D3D12Bridge_nativeHello(
    env: JNIEnv,
    _class: JClass,
    input: JString,
) -> JString {
    // 从Java获取字符串
    let input_str: String = env
        .get_string(&input)
        .expect("Couldn't get Java string")
        .into();

    log::info!("Java said: {}", input_str);

    // 构造返回字符串
    let response = format!("Hello from Rust! You said: {}", input_str);
    env.new_string(&response).unwrap().into_inner()
}
```

### 1.5 Java 端 JNI 声明

在 `fabric/src/main/java/com/dx12/D3D12Bridge.java` 中：

```java
package com.dx12;

public class D3D12Bridge {
    static {
        // 加载动态库（Windows: wgpu_mc_jni.dll）
        System.loadLibrary("wgpu_mc_jni");
    }

    // 声明native方法
    public static native void nativeInit();
    public static native String nativeHello(String input);

    // 供其他类调用的封装
    public static void init() {
        nativeInit();
    }

    public static String sayHello(String msg) {
        return nativeHello(msg);
    }
}
```

    dependsOn 'copyRustLib'

### 2.2 实现 WmRenderer 核心结构

参考wgpu-mc架构，`WmRenderer`是总协调器：

```rust
// rust/wgpu-mc/src/lib.rs
use wgpu::*;

pub struct WmRenderer {
    surface: Surface<'static>,
    device: Device,
    queue: Queue,
    config: SurfaceConfiguration,
    pub width: u32,
    pub height: u32,
}

impl WmRenderer {
    pub async fn new(window: &winit::window::Window) -> Self {
        let instance = Instance::new(InstanceDescriptor {
            backends: Backends::PRIMARY,
            ..Default::default()
        });

        let surface = instance.create_surface(window).unwrap();
        let adapter = instance
            .request_adapter(&RequestAdapterOptions {
                power_preference: PowerPreference::HighPerformance,
                compatible_surface: Some(&surface),
                ..Default::default()
            })
            .await
            .unwrap();

        let (device, queue) = adapter
            .request_device(
                &DeviceDescriptor {
                    required_features: Features::empty(),
                    required_limits: Limits::default(),
                    label: None,
                },
                None,
            )
            .await
            .unwrap();

        let size = window.inner_size();
        let config = SurfaceConfiguration {
            usage: TextureUsages::RENDER_ATTACHMENT,
            format: surface.get_supported_formats(&adapter)[0],
            width: size.width,
            height: size.height,
            present_mode: PresentMode::Fifo,  // 垂直同步，稳定
            alpha_mode: CompositeAlphaMode::Auto,
            view_formats: vec![],
            desired_maximum_frame_latency: 2,
        };
        surface.configure(&device, &config);

        Self {
            surface,
            device,
            queue,
            config,
            width: size.width,
            height: size.height,
        }
    }

    pub fn resize(&mut self, width: u32, height: u32) {
        self.width = width;
        self.height = height;
        self.config.width = width;
        self.config.height = height;
        self.surface.configure(&self.device, &self.config);
    }

    /// 渲染一帧（清屏为红色）
    pub fn render_frame(&mut self) -> Result<(), SurfaceError> {
        let output = self.surface.get_current_texture()?;
        let view = output.texture.create_view(&TextureViewDescriptor::default());

        let mut encoder = self.device.create_command_encoder(
            &CommandEncoderDescriptor { label: Some("Render Encoder") }
        );

        {
            let mut render_pass = encoder.begin_render_pass(&RenderPassDescriptor {
                label: Some("Clear Pass"),
                color_attachments: &[Some(RenderPassColorAttachment {
                    view: &view,
                    resolve_target: None,
                    ops: Operations {
                        load: LoadOp::Clear(Color { r: 0.8, g: 0.0, b: 0.0, a: 1.0 }),
                        store: StoreOp::Store,
                    },
                })],
                depth_stencil_attachment: None,
                timestamp_writes: None,
                occlusion_query_set: None,
            });
        }

        self.queue.submit(std::iter::once(encoder.finish()));
        output.present();

        Ok(())
    }
}
```

### 2.3 独立测试程序

创建 `rust/wgpu-mc/examples/simple.rs`：

```rust
use wgpu_mc::WmRenderer;
use winit::event_loop::EventLoop;

#[tokio::main]
async fn main() {
    env_logger::init();
    
    let event_loop = EventLoop::new().unwrap();
    let window = winit::window::WindowBuilder::new()
        .with_title("wgpu-mc Test")
        .with_inner_size(winit::dpi::LogicalSize::new(800, 600))
        .build(&event_loop)
        .unwrap();

    let mut renderer = WmRenderer::new(&window).await;

    event_loop.run(move |event, elwt| {
        match event {
            winit::event::Event::WindowEvent { event, .. } => match event {
                winit::event::WindowEvent::Resized(size) => {
                    renderer.resize(size.width, size.height);
                }
                winit::event::WindowEvent::CloseRequested => {
                    elwt.exit();
                }
                _ => {}
            },
            winit::event::Event::AboutToWait => {
                match renderer.render_frame() {
                    Ok(_) => {}
                    Err(wgpu::SurfaceError::Lost) => {
                        renderer.resize(renderer.width, renderer.height);
                    }
                    Err(wgpu::SurfaceError::OutOfMemory) => {
                        elwt.exit();
                    }
                    _ => {}
                }
            }
            _ => {}
        }
    }).unwrap();
}
```

## 第三阶段：Minecraft Mixin 集成（3-4周）

**目标**：用Fabric Mixin拦截游戏渲染，让Minecraft使用wgpu渲染器。

### 3.1 架构设计参考

wgpu-mc的三层架构：
- **Java层**：Fabric Mixin拦截渲染调用，通过JNI传递数据
- **JNI桥接层**：Rust实现的native方法，接收Java数据
- **Rust渲染引擎**：实际执行渲染

### 3.2 Mixin拦截 `LevelRenderer.renderLevel()`

参考你现有的实现，但调用Rust端：

```java
// fabric/src/main/java/com/dx12/mixin/LevelRendererMixin.java
package com.dx12.mixin;

import com.dx12.D3D12Bridge;
import net.minecraft.client.render.LevelRenderer;
import org.spongepowered.asm.mixin.Mixin;
import org.spongepowered.asm.mixin.injection.At;
import org.spongepowered.asm.mixin.injection.Inject;
import org.spongepowered.asm.mixin.injection.callback.CallbackInfo;

@Mixin(LevelRenderer.class)
public class LevelRendererMixin {
    @Inject(method = "renderLevel", at = @At("HEAD"), cancellable = true)
    private void onRenderLevel(
        Camera camera,
        float tickDelta,
        long limitTime,
        boolean renderBlockOutline,
        CallbackInfo ci
    ) {
        // 取消OpenGL渲染
        ci.cancel();

        // 调用Rust端渲染
        D3D12Bridge.renderFullFrame();
    }
}
```

## 📌 关键提醒

1. **项目成熟度**：wgpu-mc目前处于Beta阶段，部分功能仍在开发中，需要有一定解决"未完成"问题的能力。

2. **Minecraft版本**：当前基于快照版本25w17a开发，未来可能需要适配更新。建议关注上游项目进展。

3. **学习曲线**：Rust和wgpu都有一定门槛，但一旦掌握，维护成本远低于手写C++/DX12。

4. **与现有项目的差异**：你的DX12项目核心问题是TDR（负载过重），而wgpu方案通过WebGPU抽象层和更高效的数据传输设计，从架构层面规避了这类问题。
